#include "HX711.h"

// HX711 bağlantıları
#define DOUT 3
#define CLK 2

HX711 scale;

// Kalibrasyon faktörü
float calibration_factor = -418210;

// =================================================
// SETUP
// =================================================
void setup() {
  Serial.begin(9600);

  Serial.println("HX711 Kalibrasyonu");
  Serial.println("Tartidan tum agirliklari kaldirin.");
  Serial.println("Okumalar basladiktan sonra bilinen bir agirlik koyun.");
  Serial.println("Kalibrasyon faktorunu artirmak icin: a, s, d, f");
  Serial.println("Kalibrasyon faktorunu azaltmak icin: z, x, c, v");
  Serial.println("Tare / sifirlama icin: t");

  scale.begin(DOUT, CLK);

  scale.set_scale();
  scale.tare();

  long zero_factor = scale.read_average();

  Serial.print("Zero factor: ");
  Serial.println(zero_factor);
}

// =================================================
// LOOP
// =================================================
void loop() {
  scale.set_scale(calibration_factor);

  Serial.print("Okunan deger: ");
  Serial.print(scale.get_units(), 3);
  Serial.print(" kg");

  Serial.print(" | calibration_factor: ");
  Serial.println(calibration_factor);

  if (Serial.available()) {
    char temp = Serial.read();

    if (temp == '+' || temp == 'a') {
      calibration_factor += 10;
    }
    else if (temp == '-' || temp == 'z') {
      calibration_factor -= 10;
    }
    else if (temp == 's') {
      calibration_factor += 100;
    }
    else if (temp == 'x') {
      calibration_factor -= 100;
    }
    else if (temp == 'd') {
      calibration_factor += 1000;
    }
    else if (temp == 'c') {
      calibration_factor -= 1000;
    }
    else if (temp == 'f') {
      calibration_factor += 10000;
    }
    else if (temp == 'v') {
      calibration_factor -= 10000;
    }
    else if (temp == 't') {
      scale.tare();
      Serial.println("Tare yapildi. Terazi sifirlandi.");
    }
  }

  delay(500);
}
